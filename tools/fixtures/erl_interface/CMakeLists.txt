cmake_minimum_required(VERSION 3.25)
project(otp_ei_test_fixtures LANGUAGES C)
if(NOT IS_DIRECTORY "${OTP_PREFIX}/usr/include" OR NOT IS_DIRECTORY "${SUITE_DATA}")
    message(FATAL_ERROR "Select an installed OTP prefix and copied Erl_Interface suite data")
endif()
find_library(EI_LIBRARY NAMES ei PATHS "${OTP_PREFIX}/usr/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
set(_common "${SUITE_DATA}/../all_SUITE_data")
if(SUITE STREQUAL port_call_SUITE)
    add_library(port_call_drv MODULE "${SUITE_DATA}/port_call_drv.c")
    target_include_directories(port_call_drv PRIVATE "${OTP_PREFIX}/usr/include")
    target_link_libraries(port_call_drv PRIVATE "${EI_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
    if(NOT WIN32)
        target_link_libraries(port_call_drv PRIVATE m)
    endif()
    if(APPLE)
        target_link_options(port_call_drv PRIVATE "LINKER:-undefined,dynamic_lookup")
    endif()
    set_target_properties(port_call_drv PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${SUITE_DATA}")
else()
    string(REPLACE "_SUITE" "_test" _program "${SUITE}")
    add_executable(${_program} "${SUITE_DATA}/${_program}.c"
        "${SUITE_DATA}/${_program}_decl.c" "${_common}/ei_runner.c" "${_common}/my_ussi.c")
    set(_programs ${_program})
    if(SUITE STREQUAL ei_connect_SUITE)
        add_executable(einode "${SUITE_DATA}/einode.c")
        list(APPEND _programs einode)
    elseif(SUITE STREQUAL ei_accept_SUITE)
        add_executable(eiaccnode "${SUITE_DATA}/eiaccnode.c" "${_common}/my_ussi.c")
        list(APPEND _programs eiaccnode)
    endif()
    foreach(_target IN LISTS _programs)
        target_include_directories(${_target} PRIVATE "${OTP_PREFIX}/usr/include" "${_common}")
        target_compile_definitions(${_target} PRIVATE _GNU_SOURCE _REENTRANT)
        target_link_libraries(${_target} PRIVATE "${EI_LIBRARY}" Threads::Threads)
        if(WIN32)
            target_compile_definitions(${_target} PRIVATE __WIN32__)
            target_link_libraries(${_target} PRIVATE ws2_32)
        else()
            target_link_libraries(${_target} PRIVATE m)
        endif()
        set_target_properties(${_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${SUITE_DATA}")
    endforeach()
endif()
