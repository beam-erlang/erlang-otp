cmake_minimum_required(VERSION 3.25)
project(otp_native_test_fixtures LANGUAGES C)
if(NOT IS_DIRECTORY "${OTP_PREFIX}/usr/include" OR NOT IS_DIRECTORY "${SUITE_DATA}")
    message(FATAL_ERROR "Select an installed OTP prefix and copied upstream suite data")
endif()
if(SUITE STREQUAL dirty_nif_SUITE)
    set(_fixtures dirty_nif_SUITE on_halt_a on_halt_b on_halt_c on_halt_d on_halt_e on_halt_f echo_drv)
elseif(SUITE STREQUAL dbg_SUITE)
    set(_fixtures dbg_SUITE)
else()
    message(FATAL_ERROR "No native fixture recipe for ${SUITE}")
endif()
# This is the retained suite's complete native output inventory. The shared
# on_halt_nif.c implementation is included by each on_halt variant unchanged.
foreach(_name IN LISTS _fixtures)
    add_library(${_name} MODULE "${SUITE_DATA}/${_name}.c")
    target_include_directories(${_name} PRIVATE "${OTP_PREFIX}/usr/include")
    if(APPLE)
        target_link_options(${_name} PRIVATE "LINKER:-undefined,dynamic_lookup")
    endif()
    set_target_properties(${_name} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${SUITE_DATA}")
endforeach()
