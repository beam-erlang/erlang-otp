cmake_minimum_required(VERSION 3.25)
project(otp_regex_coexist C)
find_package(BEAM CONFIG REQUIRED COMPONENTS NIF)
find_library(PCRE2_LIBRARY NAMES pcre2-8 REQUIRED)
find_path(PCRE2_INCLUDE pcre2.h REQUIRED)
add_library(otp_re_coexist MODULE stock.c)
target_include_directories(otp_re_coexist PRIVATE "${PCRE2_INCLUDE}")
target_link_libraries(otp_re_coexist PRIVATE "${PCRE2_LIBRARY}")
beam_target_nif(otp_re_coexist)
set_target_properties(otp_re_coexist PROPERTIES PREFIX "")
